SQL essentials

DDL — defining structure

CREATE TABLE employees (
  emp_id    NUMBER(10)    GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  name      VARCHAR2(100) NOT NULL,
  dept_id   NUMBER(5),
  salary    NUMBER(10,2)  DEFAULT 0,
  hire_date DATE          DEFAULT SYSDATE,
  CONSTRAINT fk_dept FOREIGN KEY (dept_id)
    REFERENCES departments(dept_id)
);

ALTER TABLE employees ADD email VARCHAR2(200);
DROP TABLE employees CASCADE CONSTRAINTS PURGE;
Oracle uses VARCHAR2 (not VARCHAR) — always prefer it. VARCHAR is reserved for future use and behaves the same today, but may not forever.

DML — manipulating data

INSERT INTO employees (name, dept_id, salary)
VALUES ('Ana Costa', 10, 5000);

UPDATE employees
SET salary = salary * 1.1
WHERE dept_id = 10;

DELETE FROM employees
WHERE hire_date < DATE '2020-01-01';

MERGE INTO target t USING source s
  ON (t.id = s.id)
  WHEN MATCHED THEN UPDATE SET t.val = s.val
  WHEN NOT MATCHED THEN INSERT (id, val) VALUES (s.id, s.val);
TRUNCATE is DDL (auto-commits, can't roll back). DELETE is DML (transactional). Choose carefully.

Oracle-specific SELECT extras

SELECT * FROM employees
WHERE ROWNUM <= 10;            -- classic top-N (applied BEFORE ORDER BY!)

SELECT * FROM employees
ORDER BY salary DESC
FETCH FIRST 10 ROWS ONLY;     -- 12c+, correct way

SELECT * FROM employees
ORDER BY salary DESC
OFFSET 10 ROWS FETCH NEXT 5 ROWS ONLY; -- pagination

Joins and subqueries

Join types

SELECT e.name, d.dept_name
FROM employees e
JOIN departments d ON e.dept_id = d.dept_id;   -- INNER JOIN

SELECT e.name, d.dept_name
FROM employees e
LEFT JOIN departments d ON e.dept_id = d.dept_id; -- keeps all employees

SELECT e.name, d.dept_name
FROM employees e, departments d               -- Oracle (+) syntax
WHERE e.dept_id = d.dept_id(+);               -- (+) = optional side
Prefer ANSI JOIN syntax (JOIN ... ON) over the old Oracle comma syntax — it's clearer and easier to maintain.

Subqueries and WITH (CTE)

-- Correlated subquery
SELECT name, salary
FROM employees e
WHERE salary > (SELECT AVG(salary)
                FROM employees
                WHERE dept_id = e.dept_id);

-- CTE (WITH clause) — much more readable
WITH dept_avg AS (
  SELECT dept_id, AVG(salary) avg_sal
  FROM employees GROUP BY dept_id
)
SELECT e.name, e.salary, d.avg_sal
FROM employees e
JOIN dept_avg d ON e.dept_id = d.dept_id
WHERE e.salary > d.avg_sal;

Analytic (window) functions

SELECT name, dept_id, salary,
  RANK()        OVER (PARTITION BY dept_id ORDER BY salary DESC) rnk,
  DENSE_RANK()  OVER (PARTITION BY dept_id ORDER BY salary DESC) drnk,
  ROW_NUMBER()  OVER (ORDER BY salary DESC) rn,
  LAG(salary)   OVER (ORDER BY hire_date) prev_sal,
  SUM(salary)   OVER (PARTITION BY dept_id) dept_total
FROM employees;
RANK skips numbers on ties (1,1,3). DENSE_RANK does not (1,1,2). ROW_NUMBER never ties.

Built-in functions

String functions

SUBSTR('Hello World', 1, 5)     -- 'Hello'  (1-based!)
INSTR('Hello World', 'o')       -- 5
LENGTH('Hello')                -- 5
UPPER('hello')                 -- 'HELLO'
TRIM('  hello  ')             -- 'hello'
REPLACE('abc', 'b', 'X')      -- 'aXc'
REGEXP_LIKE(email, '^[A-Z]', 'i') -- regex match
LISTAGG(name, ', ') WITHIN GROUP (ORDER BY name) -- string agg

Date and number functions

SYSDATE                        -- current date+time (DB server)
SYSTIMESTAMP                   -- with timezone
TRUNC(SYSDATE)                 -- midnight today
ADD_MONTHS(SYSDATE, 3)         -- 3 months from now
MONTHS_BETWEEN(d1, d2)         -- fractional months
TO_DATE('2024-01-15', 'YYYY-MM-DD')
TO_CHAR(SYSDATE, 'DD/MM/YYYY HH24:MI')

ROUND(123.456, 2)              -- 123.46
TRUNC(123.456, 2)              -- 123.45 (no rounding)
MOD(10, 3)                     -- 1
NVL(salary, 0)                 -- replace NULL
NVL2(col, val_if_nn, val_if_n)  -- ternary null check
COALESCE(a, b, c)               -- first non-null
DECODE(status, 'A', 'Active', 'Inactive') -- Oracle switch

Aggregates and GROUP BY

SELECT dept_id,
  COUNT(*),
  SUM(salary),
  AVG(salary),
  MAX(salary),
  MIN(hire_date)
FROM employees
GROUP BY dept_id
HAVING COUNT(*) > 5
ORDER BY SUM(salary) DESC;
WHERE filters rows before grouping. HAVING filters groups after aggregation. Never use an aggregate in a WHERE clause.

PL/SQL

Block structure

DECLARE
  v_name   employees.name%TYPE;       -- anchored type
  v_count  PLS_INTEGER := 0;
  v_rec    employees%ROWTYPE;          -- whole row
BEGIN
  SELECT name INTO v_name
  FROM employees WHERE emp_id = 1;

  DBMS_OUTPUT.PUT_LINE('Name: ' || v_name);
EXCEPTION
  WHEN NO_DATA_FOUND THEN
    DBMS_OUTPUT.PUT_LINE('Not found');
  WHEN OTHERS THEN
    DBMS_OUTPUT.PUT_LINE(SQLERRM);
END;

Cursors and loops

-- Implicit cursor (simplest)
FOR rec IN (SELECT * FROM employees WHERE dept_id = 10) LOOP
  DBMS_OUTPUT.PUT_LINE(rec.name);
END LOOP;

-- Explicit cursor (for more control)
CURSOR c_emp IS SELECT * FROM employees;
OPEN c_emp;
FETCH c_emp INTO v_rec;
CLOSE c_emp;
Prefer implicit cursor FOR loops — Oracle handles OPEN, FETCH, and CLOSE automatically, and they're less error-prone.

Stored procedures and functions

CREATE OR REPLACE PROCEDURE raise_salary(
  p_emp_id IN  employees.emp_id%TYPE,
  p_pct    IN  NUMBER
) AS BEGIN
  UPDATE employees
  SET salary = salary * (1 + p_pct/100)
  WHERE emp_id = p_emp_id;
  COMMIT;
END;

CREATE OR REPLACE FUNCTION get_dept_name(
  p_id IN NUMBER
) RETURN VARCHAR2 AS
  v_name VARCHAR2(100);
BEGIN
  SELECT dept_name INTO v_name
  FROM departments WHERE dept_id = p_id;
  RETURN v_name;
END;

Triggers

CREATE OR REPLACE TRIGGER trg_audit_salary
BEFORE UPDATE OF salary ON employees
FOR EACH ROW BEGIN
  INSERT INTO salary_audit
    (emp_id, old_sal, new_sal, changed_at)
  VALUES
    (:OLD.emp_id, :OLD.salary, :NEW.salary, SYSDATE);
END;
Avoid complex logic in triggers — they fire silently and make debugging hard. Prefer procedures called explicitly.

Indexes

Index types

B-Tree (default)

Best for high-cardinality columns and equality/range queries. Created automatically on primary keys.

Bitmap

Best for low-cardinality columns (status, gender). Great for data warehouses, bad for OLTP (lock contention).

Function-based

Index on an expression: UPPER(name). Query must use the same expression to benefit.

Composite

Index on multiple columns. Column order matters — leading column must appear in the WHERE clause.

Creating indexes

CREATE INDEX idx_emp_name   ON employees(name);
CREATE UNIQUE INDEX idx_email ON employees(email);
CREATE INDEX idx_upper_name  ON employees(UPPER(name));
CREATE INDEX idx_comp        ON employees(dept_id, hire_date);
CREATE BITMAP INDEX idx_status ON employees(status);
Indexes speed up reads but slow down writes. Don't index every column — profile first with EXPLAIN PLAN.

When indexes are NOT used

Function applied to indexed column: UPPER(name) = 'ANA' without function-based index Leading wildcard: LIKE '%ana' Implicit type conversion: number col = '123' IS NULL (B-Tree doesn't store NULLs) NOT IN / != on indexed column Query returns too many rows (full scan is cheaper)

Transactions and concurrency

ACID and transaction control

BEGIN   -- Oracle starts a transaction implicitly on first DML

  INSERT INTO orders ...;
  SAVEPOINT sp1;            -- partial rollback point
  UPDATE  inventory ...;
  ROLLBACK TO sp1;          -- undo back to savepoint
  COMMIT;                   -- make all changes permanent

EXCEPTION
  WHEN OTHERS THEN
    ROLLBACK;               -- undo everything
    RAISE;
END;
Oracle uses multi-version concurrency (MVCC) — readers never block writers. Each query sees a consistent snapshot of data.

Locking

-- Explicit row lock before update
SELECT * FROM employees
WHERE emp_id = 1
FOR UPDATE;                 -- blocks others from updating

FOR UPDATE SKIP LOCKED;     -- skip already-locked rows (queue pattern)
FOR UPDATE NOWAIT;          -- fail immediately if locked
Deadlocks happen when two sessions wait on each other's locks. Oracle detects and resolves them automatically by rolling back one session.

Isolation levels

READ COMMITTED (default)

Each statement sees only committed data. Most common.

SERIALIZABLE

Entire transaction sees a consistent snapshot from its start. Can raise ORA-08177.

Database objects

Views, sequences, and synonyms

-- View: stored query
CREATE OR REPLACE VIEW v_active_emp AS
  SELECT * FROM employees WHERE status = 'A';

-- Sequence: auto-increment values
CREATE SEQUENCE emp_seq
  START WITH 1 INCREMENT BY 1 NOCACHE NOCYCLE;
emp_seq.NEXTVAL  -- get next value
emp_seq.CURRVAL  -- current value (within session)

-- Synonym: alias for an object
CREATE SYNONYM emp FOR hr.employees;

Packages

-- Package spec (public API)
CREATE OR REPLACE PACKAGE emp_pkg AS
  PROCEDURE hire(p_name VARCHAR2, p_dept NUMBER);
  FUNCTION  get_count(p_dept NUMBER) RETURN NUMBER;
END emp_pkg;

-- Package body (implementation)
CREATE OR REPLACE PACKAGE BODY emp_pkg AS
  PROCEDURE hire(p_name VARCHAR2, p_dept NUMBER) IS BEGIN
    INSERT INTO employees(name, dept_id) VALUES(p_name, p_dept);
  END;
  ...
END emp_pkg;
Group related procedures and functions into packages — they load into memory once, share state within a session, and are the standard for Oracle application code.

Other objects to know

Materialized view — cached query result, refreshed on schedule Partition — split large tables by range, list, or hash Database link (DBLINK) — query another Oracle DB Directory — OS path alias for file I/O (UTL_FILE) Job (DBMS_SCHEDULER) — scheduled PL/SQL

Performance and tuning

EXPLAIN PLAN

EXPLAIN PLAN FOR
  SELECT * FROM employees WHERE dept_id = 10;

SELECT * FROM TABLE(DBMS_XPLAN.DISPLAY());
-- or for a running session:
SELECT * FROM TABLE(DBMS_XPLAN.DISPLAY_CURSOR(FORMAT=>'ALLSTATS LAST'));
Look for FULL TABLE SCAN on large tables and high Cost values. An index range scan is usually what you want.

Hints

SELECT /*+ INDEX(e idx_emp_name) */ *
FROM employees e WHERE name = 'Ana';

SELECT /*+ FULL(e) */ * FROM employees e;  -- force full scan
SELECT /*+ USE_NL(e d) */                  -- nested loops join
SELECT /*+ PARALLEL(e, 4) */              -- parallel execution
Hints are a last resort — fix statistics and indexes first. Hints hard-code plans and break when data distribution changes.

Key tuning concepts

DBMS_STATS.GATHER_TABLE_STATS — keep optimizer stats fresh Bind variables — prevent hard parsing, reuse execution plans Bulk collect + FORALL — batch DML, avoids row-by-row context switching AWR report — workload repository, find top SQL by elapsed time V$SQL — inspect cached SQL and execution stats Result cache — cache repeated query results in SGA 
-- Bulk collect example (much faster than row-by-row)
DECLARE
  TYPE t_ids IS TABLE OF employees.emp_id%TYPE;
  l_ids t_ids;
BEGIN
  SELECT emp_id BULK COLLECT INTO l_ids
  FROM employees WHERE dept_id = 10;

  FORALL i IN l_ids.FIRST..l_ids.LAST
    UPDATE employees SET salary = salary * 1.1
    WHERE emp_id = l_ids(i);
END;
